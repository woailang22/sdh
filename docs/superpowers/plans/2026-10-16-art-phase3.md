```ts
const RARITY_FRAME_ID: Record<number, string> = {
  1: 'rarity-common',
  2: 'rarity-uncommon',
  3: 'rarity-rare',
  4: 'rarity-epic',
  5: 'rarity-mythic',
};

export interface FrameBgStyle {
  backgroundImage: string;
  backgroundSize: string;
  backgroundPosition: string;
  width: number;
  height: number;
}

/** 稀有度框背景样式（background-position 精灵，居中）；越界返回 null（调用方保留现有边框）。 */
export function rarityFrameStyle(rarity: number | undefined, size: number): FrameBgStyle | null {
  if (!Number.isInteger(rarity) || (rarity as number) < 1 || (rarity as number) > 5) return null;
  const frame = RARITY_FRAME_ID[rarity as number];
  const r = frameContent('ui-chrome.png', frame);
  if (!r) return null;
  const scale = size / 1254;
  const w = r.width * scale;
  const h = r.height * scale;
  const dx = (size - w) / 2;
  const dy = (size - h) / 2;
  return {
    backgroundImage: `url(${ART_BASE}/ui-chrome.png)`,
    backgroundSize: `${1254 * scale}px ${1254 * scale}px`,
    backgroundPosition: `${dx - r.x * scale}px ${dy - r.y * scale}px`,
    width: size,
    height: size,
  };
}
```

- [ ] **Step 1: Create component**

```tsx
import React from 'react';
import { rarityFrameStyle } from '../game/art';

interface RarityFrameProps {
  rarity?: number;
  size?: number;
  className?: string;
  children?: React.ReactNode;
}

/** 稀有度框：ui-chrome 精灵做背景，图缺失时透明（现有边框兜底）。 */
export const RarityFrame: React.FC<RarityFrameProps> = ({ rarity, size = 44, className, children }) => {
  const st = rarityFrameStyle(rarity, size);
  return (
    <div
      className={className}
      style={{ width: size, height: size, position: 'relative', display: 'flex', alignItems: 'center', justifyContent: 'center' }}
    >
      {st && (
        <div
          style={{
            position: 'absolute', inset: 0, pointerEvents: 'none', backgroundRepeat: 'no-repeat',
            backgroundImage: st.backgroundImage, backgroundSize: st.backgroundSize, backgroundPosition: st.backgroundPosition,
          }}
        />
      )}
      <div style={{ position: 'relative' }}>{children}</div>
    </div>
  );
};
```
