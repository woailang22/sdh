// server/db.ts
import Database from 'better-sqlite3';

export interface DbHandle {
  saveBuilding(b: any): void;
  listBuildings(): any[];
  savePlayer(p: any): void;
  getPlayer(id: string): any | null;
}

export function createDb(path: string): DbHandle {
  const db = new Database(path);
  db.exec(`CREATE TABLE IF NOT EXISTS buildings (id TEXT PRIMARY KEY, data TEXT NOT NULL);
    CREATE TABLE IF NOT EXISTS players (id TEXT PRIMARY KEY, data TEXT NOT NULL);
    CREATE TABLE IF NOT EXISTS meta (k TEXT PRIMARY KEY, v TEXT NOT NULL);`);
  const upsertB = db.prepare(`INSERT INTO buildings (id, data) VALUES (@id, @data) ON CONFLICT(id) DO UPDATE SET data=excluded.data`);
  const allB = db.prepare(`SELECT data FROM buildings`);
  const upsertP = db.prepare(`INSERT INTO players (id, data) VALUES (@id, @data) ON CONFLICT(id) DO UPDATE SET data=excluded.data`);
  const oneP = db.prepare(`SELECT data FROM players WHERE id = ?`);
  return {
    saveBuilding(b: any) { upsertB.run({ id: b.id, data: JSON.stringify(b) }); },
    listBuildings() { return (allB.all() as any[]).map((r) => JSON.parse(r.data)); },
    savePlayer(p: any) { upsertP.run({ id: p.id, data: JSON.stringify(p) }); },
    getPlayer(id: string) { const r = oneP.get(id) as any; return r ? JSON.parse(r.data) : null; },
  };
}
```

// src/game/death.ts
export interface DeathDrop { droppedIndexes: number[]; droppedCount: number; keepEquipment: boolean; clearReserveAmmo: boolean; }
const CAMP = ['camp_plaza', 'camp_workshop', 'camp_outskirts'];
export function calcDeathDrop(slots: unknown[], mapId: string): DeathDrop {
  if (CAMP.includes(String(mapId))) {
    return { droppedIndexes: [], droppedCount: 0, keepEquipment: true, clearReserveAmmo: false };
  }
  const filled = slots.map((s, i) => (s ? i : -1)).filter((i) => i >= 0);
  const n = Math.ceil(filled.length * 0.2);
  const picked = [...filled].sort((a, b) => a - b).slice(0, n);
  return { droppedIndexes: picked, droppedCount: picked.length, keepEquipment: true, clearReserveAmmo: true };
}
```

// src/game/daily.ts
export function dayKey(d: Date): string {
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}
export function isNextDay(last: string, today: string): boolean {
  return last !== today;
}
export interface DailyState { lastSignIn: string; kills: number; gathers: number; raids: number; claimed: boolean }
export function freshDaily(today: string): DailyState {
  return { lastSignIn: today, kills: 0, gathers: 0, raids: 0, claimed: false };
}
